//fetch with JWT.
async function getStudentProfile() {
    // 1. Lấy studentId và jwtToken từ localStorage
    const studentId = localStorage.getItem('studentId');
    const jwtToken = localStorage.getItem('jwtToken');

    // Kiểm tra xem token có tồn tại không trước khi gọi API
    if (!jwtToken) {
        console.error("Không tìm thấy token, vui lòng đăng nhập lại!");
        window.location.href = '/login.html'; // Điều hướng về trang đăng nhập
        return;
    }

    try {
        // 2. Thực hiện gọi fetch với Header Authorization
        const response = await fetch(`http://localhost:8080/api/v1/students/${studentId}`, {
            method: 'GET',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': 'Bearer ' + jwtToken // Đính kèm JWT token vào đây
            }
        });

        // 3. Xử lý kết quả trả về từ Server
        if (response.ok) {
            const studentData = await response.json();
            console.log("Lấy thông tin sinh viên thành công:", studentData);
            // Hiển thị dữ liệu lên giao diện HTML tại đây
        } else if (response.status === 401) {
            // Trường hợp Token hết hạn hoặc không hợp lệ
            console.warn("Token đã hết hạn hoặc không hợp lệ (401). Đang chuyển hướng...");
            localStorage.clear(); // Xóa sạch dữ liệu cũ
            window.location.href = '/login.html';
        } else {
            console.error("Lỗi khi gọi API, mã lỗi:", response.status);
        }

    } catch (error) {
        console.error("Lỗi kết nối mạng:", error);
    }
}

// Gọi hàm khi cần lấy thông tin
